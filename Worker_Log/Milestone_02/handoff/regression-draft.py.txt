def global_case(kind='rbfe', *, name='physical_k', value=2.):
    physical, transfer, schedule, restraints, snapshot = atom_case(kind)
    system = physical_system(physical)
    force = mm.CustomExternalForce('.5*'+name+'*(x*x+y*y+z*z)')
    force.addGlobalParameter(name, value)
    force.addParticle(physical.real_to_final['protein'], [])
    force.setName('regression:physical-global-harmonic')
    system.addForce(force)
    physical = seal_physical(system, physical.topology, ml_atom_ids=physical.ml_atom_ids,
                             old_to_new=physical.old_to_new, manifest=physical.manifest)
    return physical, resolve_protocol(physical, transfer.protocol), schedule, restraints, snapshot


def holder(stage, physical, transfer, schedule, restraints):
    if stage == 'physical':
        return PhysicalEvaluator(physical, REFERENCE)
    if stage == 'preparation':
        return build_preparation(physical, REFERENCE)
    if stage == 'native':
        return AtmEvaluator(build_atm(physical, transfer, schedule, restraints), REFERENCE)
    return build_atom(physical, transfer, schedule, restraints, REFERENCE)


@pytest.mark.parametrize('kind', ('abfe', 'rbfe'))
@pytest.mark.parametrize('stage', ('native', 'upstream'))
@pytest.mark.parametrize('admission', ('sealer', 'artifact'))
def test_actual_expression_must_match_schedule(kind, stage, admission):
    physical, transfer, schedule, restraints, snapshot = atom_case(kind)
    if stage == 'native':
        bundle = build_atm(physical, transfer, schedule, restraints)
    else:
        with build_atom(physical, transfer, schedule, restraints, REFERENCE) as run:
            bundle = run.bundle
    system = mm.XmlSerializer.deserialize(bundle.system_xml)
    force = next(f for f in system.getForces() if isinstance(f, mm.ATMForce))
    force.setEnergyFunction('u0')
    with pytest.raises(IdentityError, match='expression|schedule'):
        if admission == 'sealer':
            seal_alchemical(system, physical, transfer, schedule, restraints)
        else:
            xml = mm.XmlSerializer.serialize(system)
            altered = replace(bundle, system_xml=xml,
                              system_sha256=hashlib.sha256(xml.encode()).hexdigest())
            with AtmEvaluator(altered, REFERENCE) as evaluator:
                evaluator.evaluate(snapshot, 'first')

@pytest.mark.parametrize('stage', ('physical', 'preparation', 'native', 'upstream'))
@pytest.mark.parametrize('restored', (False, True))
def test_fixed_physical_globals_reject_context_and_state_mutation(stage, restored):
    physical, transfer, schedule, restraints, snapshot = global_case()
    with holder(stage, physical, transfer, schedule, restraints) as run:
        evaluator = run.evaluator if stage == 'upstream' else run
        args = (snapshot,) if stage in ('physical', 'preparation') else (snapshot, 'first')
        evaluator.evaluate(*args)
        evaluator.context.setParameter('physical_k', 200.)
        if restored:
            state = evaluator.context.getState(getPositions=True, getParameters=True)
            evaluator.context.setParameter('physical_k', 2.)
            evaluator.context.setState(state)
        with pytest.raises(IdentityError, match='physical_k|parameter'):
            evaluator.evaluate(*args)


@pytest.mark.parametrize('stage', ('physical', 'preparation', 'native', 'upstream'))
def test_fixed_physical_global_positive_control(stage):
    physical, transfer, schedule, restraints, snapshot = global_case()
    coordinates = np.asarray(snapshot.positions_nm)
    extra_energy = float(np.dot(coordinates[4], coordinates[4]))
    extra_forces = np.zeros((7, 3))
    extra_forces[4] = -2.*coordinates[4]
    with holder(stage, physical, transfer, schedule, restraints) as run:
        evaluator = run.evaluator if stage == 'upstream' else run
        if stage in ('physical', 'preparation'):
            energy, forces = physical_answer(coordinates)
            actual = evaluator.evaluate(snapshot)
            assert actual.energy_kj_mol == pytest.approx(energy+extra_energy, abs=1e-8, rel=0)
            np.testing.assert_allclose(actual.forces_kj_mol_nm, forces+extra_forces, atol=1e-7, rtol=0)
        else:
            for state in schedule.states:
                u0, f0 = physical_answer(mapped_positions(coordinates, 'rbfe', 0))
                u1, f1 = physical_answer(mapped_positions(coordinates, 'rbfe', 1))
                expression, weights, soft = nonlinear_answer(u0+extra_energy, u1+extra_energy, state.parameters)
                outside, fk = outside_answer(coordinates)
                check(evaluator.evaluate(snapshot, state.state_id),
                      (u0+extra_energy, u1+extra_energy, expression, outside,
                       weights[0]*f0+weights[1]*f1+extra_forces+fk))
                assert evaluator.context.getParameter('physical_k') == 2.


@pytest.mark.parametrize('kind', ('abfe', 'rbfe'))
@pytest.mark.parametrize('stage', ('native', 'upstream'))
def test_schedule_globals_cannot_control_physical_children(kind, stage):
    physical, transfer, _, restraints, snapshot = global_case(kind, name='Lambda1', value=0.)
    schedule = production_schedule((('first', production_parameters(Lambda1=0.)),
                                    ('second', production_parameters(Lambda1=.3))))
    with pytest.raises(IdentityError, match='collision|physical|parameter|schedule'):
        with holder(stage, physical, transfer, schedule, restraints) as run:
            evaluator = run.evaluator if stage == 'upstream' else run
            evaluator.evaluate(snapshot, 'first')
            evaluator.evaluate(snapshot, 'second')


@pytest.mark.parametrize('stage', ('native', 'upstream'))
def test_explicit_state_restore_preserves_physical_globals(stage):
    physical, transfer, schedule, restraints, snapshot = global_case()
    with holder(stage, physical, transfer, schedule, restraints) as run:
        evaluator = run.evaluator if stage == 'upstream' else run
        evaluator.evaluate(snapshot, 'first')
        evaluator.context.setParameter('physical_k', 200.)
        state = evaluator.context.getState(getPositions=True, getParameters=True)
        evaluator.context.setParameter('physical_k', 2.)
        evaluator.restore_state(state, 'second')
        assert evaluator.context.getParameter('physical_k') == 2.
        assert evaluator.context.getParameter('Lambda1') == .3
        evaluator.evaluate(snapshot, 'second')


@pytest.mark.parametrize('kind', ('abfe', 'rbfe'))
@pytest.mark.parametrize('stage', ('export', 'native', 'upstream'))
def test_renaming_and_regrouping_cannot_hide_duplicate_physical_content(kind, stage):
    physical, transfer, schedule, restraints, snapshot = atom_case(kind)
    system = physical_system(physical)
    duplicate = copy.copy(system.getForce(0))
    duplicate.setName('regression:renamed-duplicate')
    duplicate.setForceGroup(7)
    system.addForce(duplicate)
    physical = seal_physical(system, physical.topology, ml_atom_ids=physical.ml_atom_ids,
                             old_to_new=physical.old_to_new, manifest=physical.manifest)
    transfer = resolve_protocol(physical, transfer.protocol)
    with pytest.raises(IdentityError, match='duplicate'):
        if stage == 'export':
            export_physical(physical)
        else:
            with holder(stage, physical, transfer, schedule, restraints) as run:
                evaluator = run.evaluator if stage == 'upstream' else run
                evaluator.evaluate(snapshot, 'first')
